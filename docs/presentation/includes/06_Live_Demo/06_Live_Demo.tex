\addsectionframe{6. Interaktive Live-Demo}

\begin{itemframe}{Ablauf der 45 Minuten}
  \item Kontext und Anforderungen: 10 Minuten
  \item Architektur, Design und KI-Einsatz: 15 Minuten
  \item Qualität und Deployment: 5 Minuten
  \item Interaktive Live-Demo: 15 Minuten
\end{itemframe}

\begin{exercise}{Aufgabe an die Prüfenden}
  Öffnen Sie ein Fachthema Ihrer Wahl. Wählen Sie eine Übung oder ein Beispiel,
  verändern Sie eine Eingabe und starten Sie die Simulation. Beschreiben Sie
  anschließend, welcher Zwischenschritt die Lösung erklärt.
\end{exercise}

\begin{itemframe}{Moderation der Demo}
  \item Zwei Endgeräte oder zwei Browserfenster vorbereiten: Prüfende bedienen eines selbst.
  \item Aufgabe A: Wort in einem endlichen Automaten ändern und die Zustandsfolge prüfen.
  \item Aufgabe B: DNF- oder Zahlensystem-Aufgabe bearbeiten und Rückmeldung auslösen.
  \item Wir zeigen danach Import/Export eines Arbeitsstands und ordnen die Ansicht der Architektur zu.
\end{itemframe}

\begin{itemframe}{Demo-Absicherung}
  \item Startzustand und direkte URLs vor der Präsentation testen.
  \item Eine vorbereitete Eingabe und ein gespeicherter Arbeitsstand liegen lokal bereit.
  \item Fällt die Bedienung durch die Prüfenden aus, übernimmt ein Teammitglied denselben Ablauf.
\end{itemframe}
