\begin{itemframe}{Kodierung konfigurieren}
  \item Zwei Verfahren vollständig umgesetzt: \textbf{UTF-8} und \textbf{Base64}
  \item Freier Text oder Übungsaufgabe als Eingabe -- Ver- und Entschlüsselung unterstützt
  \item Strenge Decodierung: \textit{Overlong Encodings} werden als Sicherheitsrisiko abgelehnt, nicht druckbare Zeichen erscheinen als Escape-Sequenzen
  \item Übungen dynamisch generiert (easy = ASCII, medium = 2-Byte, hard = 3-Byte-Unicode) -- Fortschritt dabei bewusst nicht persistent
  \note{
    \textbf{Person: Enrico}

    \textbf{Zeit: 1 min}

    \begin{itemize}
      \item Einstieg: Kodierung kam später dazu, auf Wunsch des Auftraggebers --
      \item im Lastenheft war sie nicht vorgesehen.
      \item Besonderheit gegenüber anderen Modulen: Übungen sind ephemer und werden
      \item nicht dauerhaft gespeichert, der freie Simulator dagegen schon.
      \item Überleitung: "Wie sieht die Transformation im Detail aus?"
    \end{itemize}
  }
\end{itemframe}
