\addsectionframe{Fachmodul: Kodierung}

% ---------------------------------------------------------------------------
\begin{itemframe}{Kodierung konfigurieren}
  \item Zwei Verfahren: \textbf{UTF-8} und \textbf{Base64}
  \item Freier Text oder dynamische Übung als Eingabe
  \item Kodieren und Dekodieren in beide Richtungen möglich
  \item Aufklappbare Zwischenschritte mit Farbcodierung
  \item Dynamische Übungen nach Schwierigkeit (easy / medium / hard)
  \note{
    \textbf{Person: Enrico}

    \textbf{Zeit: 1 min}

    \begin{itemize}
      \item Modul kam auf Wunsch des Auftraggebers zusätzlich hinzu
      \item Eingabe wählbar: freier Text oder Übung
      \item Verfahren + Richtung (kodieren / dekodieren) wählbar
      \item Details aufklappbar: UTF-8 (Codepoints, Steuerbits, Hex) oder Base64 (6-Bit, Alphabet, Padding)
      \item Farbcodierung hilft, Bit-Muster zu erkennen
      \item Übungen dynamisch generiert, nicht persistent -- bewusst als Wegwerf-Tests
      \item Freier Simulator speichert dagegen automatisch
      \item Überleitung: "Wie läuft das konkret in der App ab?"
    \end{itemize}
  }
\end{itemframe}
