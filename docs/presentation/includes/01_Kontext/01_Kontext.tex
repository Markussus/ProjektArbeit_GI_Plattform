\addsectionframe{1. Kontext und Zielbild}

\begin{itemframe}{Ausgangslage}
  \item Themen wie Automaten, Zahlensysteme und Kodierungen sind abstrakt.
  \item Statische Aufgaben zeigen meist nur Eingabe und Ergebnis.
  \item Lernende sollen Zwischenschritte selbst verändern und beobachten können.
\end{itemframe}

\begin{itemframe}{Ziel der Projektarbeit}
  \item Eine browserbasierte Lern- und Übungsplattform für \emph{Grundlagen Informatik}.
  \item Interaktive Simulatoren erklären den Rechen- oder Ausführungsweg Schritt für Schritt.
  \item Einheitliche Oberfläche für zwei Lernbereiche und acht Fachthemen.
\end{itemframe}
