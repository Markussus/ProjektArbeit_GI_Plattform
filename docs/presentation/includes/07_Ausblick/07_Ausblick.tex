\addsectionframe{7. Ausblick und Fazit}

\begin{itemframe}{Erweiterbarkeit und Perspektiven}
  \item \textbf{Lernbereiche}: Neue Module lassen sich zentral im \texttt{moduleCatalog} ohne Umbauten registrieren.
  \item \textbf{Aufgabenverwaltung}: Zukünftige Erstellung und Variantenbildung von Aufgaben direkt über die Oberfläche.
  \item \textbf{Lernunterstützung}: PDF-Arbeitsblattexport für Präsenzveranstaltungen und Theorie-Quizze.
  \item \textbf{Infrastruktur}: Abwägung zwischen bewusster Offline-/Backendfreiheit und optionaler LMS-/Moodle-Anbindung.
\end{itemframe}

\begin{itemframe}{Nächste sinnvolle Schritte}
  \item Echtes Feedback aus Lehrveranstaltungen dokumentieren und in Anforderungen zurückführen.
  \item Produktiv-Deployment mit Hosting, Freigabe und Rückfallplan festlegen.
  \item Mobile Optimierung für komplexe Tabellen und Zustandsdiagramme auf kleinen Bildschirmen.
\end{itemframe}

\begin{itemframe}{Fazit}
  \item Die Plattform verbindet fachliche Modelle mit bearbeitbaren, schrittweisen Simulationen.
  \item Architektur und gemeinsame UI-Bausteine halten die Fachmodule konsistent und erweiterbar.
  \item Die Live-Demo prüft den Kernanspruch unmittelbar: Die Plattform erklärt sich durch Benutzung.
\end{itemframe}
