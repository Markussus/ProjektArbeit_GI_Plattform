\begin{itemframe}{Reguläre Sprachen -- Überblick}
  \item \textbf{Rechtslineare Grammatik} selbst schreiben -- geprüft schon beim Tippen
  \item Drei Modi: \textbf{Aufzählung}, \textbf{Erzeugung}, \textbf{Worterkennung}
  \item \textbf{Übungen} mit Rückmeldung: fehlende und falsche Wörter
          \note{
            \textbf{Person: Kristian}

            \textbf{Zeit: 1 min}

            \begin{itemize}
              \item Einstieg zweites Thema, parallel zum Überblick bei den Automaten.
              \item Grammatik selbst schreiben, eine Produktion pro Zeile. Prüfung beim Tippen.
              \item Drei Modi nur benennen $\rightarrow$ Details auf der nächsten Folie.
              \item In der Oberfläche: Grammatik -> Sprache, Sprache -> Grammatik, Wort erkennen.
              \item \textbf{Übungen}: geprüft wird das Verhalten
                    $\rightarrow$ fehlende und zu viele Wörter.
              \item Überleitung: diese drei Arbeitsweisen im Zusammenhang.
            \end{itemize}
          }
\end{itemframe}

\begin{fullframe}{Reguläre Sprachen -- drei Arbeitsweisen}
  \begin{center}
    \begin{tikzpicture}[
        >=Stealth,
        box/.style={draw=thwsblue, very thick, rounded corners=3pt, fill=thwsblue!5,
                    minimum width=30mm, minimum height=12mm, align=center, font=\small},
        res/.style={draw=thwspetrol, very thick, rounded corners=3pt, fill=thwspetrol!8,
                    minimum width=26mm, minimum height=10mm, align=center, font=\small},
        lbl/.style={font=\scriptsize, thwsorange},
        fmt/.style={font=\scriptsize, text=black!60, align=center}
      ]
      \node[box] (gram) {Grammatik\\{\scriptsize rechtslinear}};
      \node[box, right=42mm of gram] (lang) {Sprach-\\beschreibung};

      \node[fmt, above=2.5mm of gram] {erzeugte Grammatik:\\ausführlich oder vereinfacht};
      \node[fmt, above=2.5mm of lang] {Wortliste oder Mengenschreibweise\\$\{a^n b \mid n \geq 1\}$};

      \draw[->, very thick, thwsorange] ([yshift=3mm]gram.east)
        -- node[lbl, above, align=center] {Aufzählung} ([yshift=3mm]lang.west);
      \draw[<-, very thick, thwsorange] ([yshift=-5mm]gram.east)
        -- node[lbl, below, align=center] {Erzeugung} ([yshift=-5mm]lang.west);

      \node[res, below=16mm of gram] (word) {Wort + Grammatik};
      \node[res, right=42mm of word] (verdict) {akzeptiert / abgelehnt\\{\scriptsize mit Begründung}};
      \draw[->, very thick, thwspetrol] (word) -- node[lbl, above, thwspetrol] {Worterkennung} (verdict);
    \end{tikzpicture}
  \end{center}
            \note{
              \textbf{Person: Kristian}

              \textbf{Zeit: 1 min}

              \begin{itemize}
                \item \textbf{Aufzählung}: Grammatik $\rightarrow$ Wörter. Begrenzt nach Länge
                      und Anzahl.
                \item \textbf{Erzeugung}: Sprache $\rightarrow$ Grammatik.
                      Strikt rechtslinear, ausführlich oder vereinfacht.
                \item Eingabeformate der Sprache: Wortliste oder Mengenschreibweise.
                \item \textbf{Worterkennung}: mehrere Wörter gegen die Grammatik,
                      mit Begründung statt nur ja/nein.
                \item Frei sagen, steht nicht auf der Folie: Modi bleiben gemountet
                      $\rightarrow$ begonnene Eingabe überlebt den Moduswechsel.
              \end{itemize}
            }
\end{fullframe}
