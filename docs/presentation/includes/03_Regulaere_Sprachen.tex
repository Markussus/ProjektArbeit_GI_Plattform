\begin{itemframe}{Reguläre Sprachen -- Überblick}
  \item \textbf{Rechtslineare Grammatik} selbst schreiben -- geprüft schon beim Tippen
  \item Drei Modi: \textbf{Aufzählung}, \textbf{Erzeugung}, \textbf{Worterkennung}
  \item \textbf{Übungen} mit Rückmeldung: fehlende und falsche Wörter

  \note{
            \textbf{Person: Kristian}

            \textbf{Zeit: 1 min}

            \begin{itemize}
              \item Die Grammatik wird Zeile für Zeile eingegeben
              und live beim Tippen geprüft.
              Ein Berechnen-Knopf fehlt bewusst: Jeder Fehler – wie eine falsche Produktion
              oder ein nicht deklariertes Symbol – wird direkt in der betroffenen Zeile gemeldet.
              \item Drei Modi nur benennen $\rightarrow$ Details auf der nächsten Folie.
              \item \textbf{Übungen}: Prüfung der erkannten Sprache durch konkrete Gegenbeispiele
              wie fehlende oder falsche Wörter. Genau wie bei den Automaten
            \end{itemize}
  }
\end{itemframe}

      \node[box] (gram) {Grammatik\\{\scriptsize rechtslinear}};
      \node[box, right=42mm of gram] (lang) {Sprach-\\beschreibung};

      \node[fmt, above=2.5mm of gram] {erzeugte Grammatik:\\ausführlich oder vereinfacht};
      \node[fmt, above=2.5mm of lang] {Wortliste oder Mengenschreibweise\\$\{a^n b \mid n \geq 1\}$};

      \draw[->, very thick, thwsorange] ([yshift=3mm]gram.east)
        -- node[lbl, above, align=center] {Aufzählung} ([yshift=3mm]lang.west);
      \draw[<-, very thick, thwsorange] ([yshift=-5mm]gram.east)
        -- node[lbl, below, align=center] {Erzeugung} ([yshift=-5mm]lang.west);

      \node[res, below=16mm of gram] (word) {Wort + Grammatik};
      \node[res, right=42mm of word] (verdict) {akzeptiert / abgelehnt\\{\scriptsize mit Begründung}};
      \draw[->, very thick, thwspetrol] (word) -- node[lbl, above, thwspetrol] {Worterkennung} (verdict);
    \end{tikzpicture}
  \end{center}

  \note{
              \textbf{Person: Kristian}

              \textbf{Zeit: 1 min}

              \begin{itemize}
                \item \textbf{Aufzählung}: Grammatik $\rightarrow$ Wörter der Sprache.
                Man kann nach Länge und Anzahl begrenzen.
                \item \textbf{Erzeugung}: Die Gegenrichtung.
                Sprache $\rightarrow$ Grammatik.
                Sprache kann man als \textbf{Wortliste} oder \textbf{Mengenschreibweise} darstellen.
                \item \textbf{Ausführlich} $\rightarrow$ jedes Präfix ein eigenes Nichtterminal bekommt.
                \item \textbf{Vereinfachte} Grammatik $\rightarrow$ Präfixe werden zusammengefasst.
                \item \textbf{Worterkennung}: Mehrere Wörter gegen die Grammatik testen.
                Ergebnis: Man bekommt Begründung statt nur ja/nein.
              \end{itemize}
  }
\end{fullframe}
