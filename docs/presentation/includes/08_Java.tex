\begin{contentsframe}{Java -- Überblick}
  \centering
  \vspace{2mm}
  \includegraphics[width=0.95\textwidth,height=0.75\textheight,keepaspectratio]{java_diagram.pdf}
  \note{
    \textbf{Person: Markus}

    \textbf{Zeit: 1 min}

    \begin{itemize}
      \item \textbf{Client-Architektur:} 100\% lokale Ausführung im Browser-Web-Worker; benötigt keinen Backend-Server.
      \item \textbf{Scanner-Substitution:} Vorbereitete Eingaben (\texttt{stdin}) werden vorab injiziert, da Web-Worker nicht synchron auf Tastatur blockieren können.
      \item \textbf{TeaVM Pipeline:} Java-Quellcode wird zu Bytecode kompiliert und via TeaVM in WebAssembly (\texttt{app.wasm}) übersetzt.
      \item \textbf{FFI \& Konsole:} Byteweiser Hook von \texttt{putchar} fängt \texttt{stdout}/\texttt{stderr} ab und flusht alle 50\,ms in die UI.
    \end{itemize}
  }
\end{contentsframe}

\begin{itemframe}{Besonderheiten und Einschränkungen}
    \item \textbf{Offline nutzbar} -- Verteilung als ZIP, keine externe Verbindung nötig
    \vspace{1em}
    \item \textbf{Kein lokales Java} -- kein JDK, Gradle oder Git erforderlich
    \vspace{1em}
    \item \textbf{Kein Java-Backend} -- Kompilierung und Ausführung lokal im Browser
    \vspace{1em}
    \item \textbf{Plattformunabhängig} -- Nutzung auch auf Tablets / iPads im Browser
    \vspace{1em}
    \item \textbf{Browser-Sandbox} -- kein interaktives \texttt{System.in}, kein Dateisystemzugriff
    \vspace{1em}
    \item \textbf{Für Lernaufgaben ausreichend} -- Code, (Eingaben) und \texttt{stdout}/\texttt{stderr}
    \note{
      \textbf{Person: Markus}

      \textbf{Zeit: 0.75 min}

      \begin{itemize}
        \item \textbf{Zero-Installation:} Kein JDK, kein Gradle, keine Pfade; läuft direkt im Browser (auch auf iPads/Tablets).
        \item \textbf{Kein Server:} 100\% offlinefähig im ZIP-Paket; keine Serverkosten, kein DSGVO-Risiko.
        \item \textbf{Sandbox \& Didaktik:} Sicherheitsisolierung (kein Dateisystem/Netzwerk) schützt den Rechner; optimal zugeschnitten auf GI-Programmieraufgaben.
      \end{itemize}
    }
\end{itemframe}
