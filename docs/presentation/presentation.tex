\begin{contentsframe}{Simulator-Architektur}
\centering
\vspace{1mm}
\resizebox{0.88\textwidth}{!}{%
\begin{tikzpicture}[node distance=9mm and 7mm]
  \node[diagramnode, minimum width=3.3cm, font=\normalsize] (data) {Konfiguration +\\Laufzeitzustand};
  \node[diagramnode, right=of data, minimum width=4.0cm, font=\normalsize] (iface) {\texttt{ISimulator<TConfig,TState>}\\gemeinsames Interface};
  \node[diagramnode, right=of iface, minimum width=3.7cm, font=\normalsize] (hook) {\texttt{useSimulationEngine}\\Steuerung + Synchronisation};
  \node[diagramnode, below=of hook, minimum width=3.7cm, font=\normalsize] (ui) {React-UI\\Play / Pause / Step / Reset};
  \node[diagramnode, below=of iface, minimum width=4.0cm, font=\normalsize] (impl) {Konkrete Simulatorlogik};

  \draw[diagramarrow] (data) -- (iface);
  \draw[diagramarrow] (iface) -- (hook);
  \draw[diagramarrow] (iface) -- (impl);
  \draw[diagramdouble] (impl) -- (hook);
  \draw[diagramdouble] (hook) -- (ui);
\end{tikzpicture}%
}
  \note{
    \textbf{Person: Richard}

    \textbf{Zeit: 2,5 min}

    \begin{itemize}
      \item \textbf{Architekturprinzip:} Strikte Trennung von formaler Rechenlogik und UI-Rendering. Fachlogik ist reines TypeScript per Vitest unit-testbar.
      \item \textbf{Gemeinsames Interface:} \texttt{ISimulator} für alle schrittbasierten Modelle (TM, EA, MM). Methoden: \texttt{initialize}, \texttt{step}, \texttt{stepBack}, \texttt{reset}, \texttt{getCurrentState}, \texttt{isFinished}.
      \item \textbf{Konkrete Simulatoren} Kapseln mathematische Übergangsfunktion ($\delta$) und Validierung.
      \item \textbf{Steuerungs-Hook \texttt{useSimulationEngine}:} Schlägt die Brücke zur reaktiven React-Welt. Kapselt Browser-Timer (\texttt{setInterval}), Ausführungsstatus (\texttt{IDLE}, \texttt{RUNNING}, \texttt{PAUSED}, \texttt{ERROR}), Geschwindigkeit (\texttt{speed}) und stoppt automatisch bei \texttt{isFinished}.
      \item \textbf{Wiederverwendbare UI (\texttt{SimulatorControls}):} Standardisierte Bedienelemente (Play, Pause, Step vor/zurück, Reset, Tempo-Slider) für alle Simulatoren.
    \end{itemize}
  }
\end{contentsframe}

\begin{itemframe}{Import, Export und lokale Persistenz}
  \item Einheitliches Transfer-Codec Interface
          \vspace{1em}
  \item Kompakte Base64URL-Transferstrings (12Bit topicStorageId + 6 Bit taskStorageId + Payload)
          \vspace{1em}
  \item Kollisionsfreie lokale Persistenz

  \note{
    \textbf{Person: Markus}

    \textbf{Zeit: 1,5 min}

    \begin{itemize}
      \item \textbf{Interface:} \texttt{ISimulatorTransferCodec<TValue>} statt Monolith. Inversion of Control: Simulator meldet Export beim Mount im Zustand-Store an; Header bleibt entkoppelt.
      \item \textbf{Base64URL:} 3 Zeichen Header (2 Chars Topic-ID, 1 Char Task-ID) fürs Routing. URL-sicher nach RFC 4648 (\texttt{-}/\texttt{\_}, kein Padding \texttt{=}). BitWriter für Automaten, Tokenizer + Deflate für Java.
      \item \textbf{Local Storage:} Zentraler Key \texttt{"TrainingCenterStorage"}. Kollisionsfrei durch 18-Bit Topic- und Task-Namespacing. Entprelltes Autosave, autom. Restore beim Laden.
    \end{itemize}
  }
  \end{itemframe}

\begin{itemframe}{Grenzen und Erkenntnisse beim KI-Einsatz}
  \item Kontextverlust bei großen Aufgaben $\rightarrow$ \texttt{handover.md}
              \vspace{1em}
  \item Duplikate und toter Code $\rightarrow$ Code Review mit KI
              \vspace{1em}
  \item Nutzen durch kleine, testbare Schritte
    \note{
      \textbf{Person: Markus}

      \textbf{Zeit: 1,5 min}

      \begin{itemize}
        \item \textbf{Kontextverlust:} LLMs verlieren bei längeren Sessions den Faden. Die Datei \texttt{handover.md} hält offene Aufgaben und Architektur-Entscheidungen verbindlich fest.
        \item \textbf{Code-Qualität:} Modelle bauen gerne eigene Hilfsfunktionen, obwohl es sie im Projekt schon gibt. Manuelle Diff-Reviews und gezielte Refactoring-Prompts verhindern das.
        \item \textbf{Arbeitsweise:} Ein großer Prompt auf einmal funktioniert fast nie. Zuverlässige Ergebnisse gab es nur in kleinen Schritten von der Schnittstelle bis zu den Tests.
        \item \textbf{Dokumentation:} Der Dozent wollte viel KI-Nutzung sehen. Wir haben damit eigene Stichpunkte ausformuliert und Texte geglättet, aber jeden Satz selbst geprüft.
      \end{itemize}
    }
  \end{itemframe}

\begin{itemframe}{Deployment}
  \item Statischer Vite-Build als ZIP ausgeliefert
  \vspace{1em}
  \item Deployment Skript für automatischen GitLab release
  \vspace{1em}
  \item Browserbetrieb durch lokalen Server
  \vspace{1em}
  \item Online hosting durch Auftraggeber abgelehnt
    \note{
      \textbf{Person: Markus}

      \textbf{Zeit: 1 min}

      \begin{itemize}
        \item \textbf{Vite-Build:} Vollständig statische Client-SPA (\texttt{dist/}) als ZIP; benötigt keinerlei Server-Runtime oder Datenbank.
        \item \textbf{GitLab-Release:} CI-Job baut, testet und paketiert bei Tags vollautomatisch; erzeugt versionierte GitLab-Releases.
        \item \textbf{Lokaler Server vs. file://:} \texttt{file://} blockiert ES-Module, dyn. Chunks und Monaco-Web-Worker (CORS/Origin null). Lokaler Mini-Server liefert zwingend benötigten HTTP-Origin.
        \item \textbf{Hosting vs. PWA:} Kein zentrales Web-Hosting (keine Serverkosten/Wartung/DSGVO). PWA scheitert offline ohne initiale HTTPS-Domain; lokale ZIP-Verteilung 100\% autark.
      \end{itemize}
    }
  \end{itemframe}
