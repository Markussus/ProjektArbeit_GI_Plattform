\begin{itemframe}{Import, Export und lokale Persistenz}
  \item Einheitliches Transfer-Codec Interface
          \vspace{1em}
  \item Kompakte Base64URL-Transferstrings (12Bit topicStorageId + 6 Bit taskStorageId + Payload)
          \vspace{1em}
  \item Kollisionsfreie lokale Persistenz
    \note{
      \textbf{Person: Markus} \hfill \textbf{Zeit: 1,5 min}

      \begin{itemize}
        \item \textbf{Interface:} \texttt{ISimulatorTransferCodec<TValue>} statt Monolith. Inversion of Control: Simulator meldet Export beim Mount im Zustand-Store an; Header bleibt entkoppelt.
        \item \textbf{Base64URL:} 3 Zeichen Header (2 Chars Topic-ID, 1 Char Task-ID) fürs Routing. URL-sicher nach RFC 4648 (\texttt{-}/\texttt{\_}, kein Padding \texttt{=}). BitWriter für Automaten, Tokenizer + Deflate für Java.
        \item \textbf{Local Storage:} Zentraler Key \texttt{"TrainingCenterStorage"}. Kollisionsfrei durch 18-Bit Topic- und Task-Namespacing. Entprelltes Autosave, autom. Restore beim Laden.
      \end{itemize}
    }
  \end{itemframe}

\begin{itemframe}{Grenzen und Erkenntnisse beim KI-Einsatz}
  \item Kontextverlust bei großen Aufgaben $\rightarrow$ \texttt{handover.md}
              \vspace{1em}
  \item Duplikate und toter Code $\rightarrow$ Code Review mit KI
              \vspace{1em}
  \item Nutzen durch kleine, testbare Schritte
    \note{
      \textbf{Person: Markus} \hfill \textbf{Zeit: 1,5 bis 2 min}

      \begin{itemize}
        \item \textbf{Kontextverlust:} LLMs verlieren bei langen Tasks den Projektüberblick. Lösung: \texttt{handover.md} sichert Entscheidungen, offenen Stand und Architekturvorgaben über Session-Grenzen hinweg.
        \item \textbf{Code-Qualität:} KI erfindet oft vorhandene Utilities neu oder erzeugt toten Code. Lösung: Striktes Diff-Review durch Entwickler und gezielte KI-gestützte Refactoring-Prompts.
        \item \textbf{Arbeitsweise:} Große One-Shot-Prompts scheitern regelmäßig. Maximaler Hebel nur durch atomare, testbare Schritte (Interface $\rightarrow$ Core-Logik $\rightarrow$ Tests $\rightarrow$ UI).
      \end{itemize}
    }
  \end{itemframe}

\begin{itemframe}{Deployment}
  \item Statischer Vite-Build als ZIP ausgeliefert
  \vspace{1em}
  \item Deployment Skript für automatischen GitLab release
  \vspace{1em}
  \item Browserbetrieb durch lokalen Server
  \vspace{1em}
  \item Online hosting durch Auftraggeber abgelehnt
    \note{
      \textbf{Person: Markus} \hfill \textbf{Zeit: 1 min}

      \begin{itemize}
        \item \textbf{Vite-Build:} Vollständig statische Client-SPA (\texttt{dist/}) als ZIP; benötigt keinerlei Server-Runtime oder Datenbank.
        \item \textbf{GitLab-Release:} CI-Job baut, testet und paketiert bei Tags vollautomatisch; erzeugt versionierte GitLab-Releases.
        \item \textbf{Lokaler Server vs. file://:} \texttt{file://} blockiert ES-Module, dyn. Chunks und Monaco-Web-Worker (CORS/Origin null). Lokaler Mini-Server liefert zwingend benötigten HTTP-Origin.
        \item \textbf{Hosting vs. PWA:} Kein zentrales Web-Hosting (keine Serverkosten/Wartung/DSGVO). PWA scheitert offline ohne initiale HTTPS-Domain; lokale ZIP-Verteilung 100\% autark.
      \end{itemize}
    }
  \end{itemframe}
